import { readFileSync } from 'node:fs';
import { basename, join } from 'node:path';
const die = (m) => { console.log('DIE', m); process.exit(2); };
const MACHINE = new Set(['/usr/bin/systemd-run', '/usr/bin/env', '/usr/bin/bwrap', '/usr/bin/systemctl']);
const src = readFileSync(new URL('./fns2.mjs', import.meta.url), 'utf8');
const { parseLog, judge } = new Function('die','basename','join','MACHINE', src + '; return { parseLog, judge };')(die,basename,join,MACHINE);
const [log, tree] = process.argv.slice(2);
const p = parseLog(readFileSync(log, 'utf8'));
const site = { id: 'AT802', kind: (e) => basename(e.path) === 'node', match: (a) => a[1]?.endsWith('/scripts/anchor-truth.mjs') && a.includes('--self') };
const j = judge(p, tree, site, 'NOPE_CANARY');
console.log(JSON.stringify(j));
const runnerFile = join(tree, 'scripts', 'cap-run.mjs');
const anc = (pid) => { const o=[pid]; let q=p.parent.get(pid); const s=new Set(o); while(q!==undefined && !s.has(q)){s.add(q);o.push(q);q=p.parent.get(q);} return o; };
const cands = p.execs.filter((e) => e.pid !== p.rootPid && !MACHINE.has(e.path) && e.argv[1] !== runnerFile && site.match(e.argv) && site.kind(e));
let shown=0;
for (const e of cands) {
  const one = judge({ ...p, execs: p.execs }, tree, { ...site, match: (a)=> a===e.argv }, 'NOPE');
  if (one.capped === one.n) continue;
  if (shown++ > 1) break;
  console.log('MISS pid', e.pid, 'seq', e.seq);
  for (const a of anc(e.pid)) console.log('  anc', a, p.execs.filter(x=>x.pid===a).map(x=>`${x.seq}:${basename(x.path)} ${x.argv.slice(1,3).join(' ')}`.slice(0,90)).join(' | '));
}
