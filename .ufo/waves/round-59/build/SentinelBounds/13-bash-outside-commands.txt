# census-identical 0-3 space id extract.
# sources-census.sh:40 and :615: 0-3 leading ASCII spaces, then '|'.
# id = trimmed second '|'-field. Keep ^S[0-9]{3,}$.

extract_ungated() {
  gawk '
    function trim(s) { gsub(/^[ \t]+|[ \t]+$/, "", s); return s }
    {
      line = $0
      if (line !~ /^ {0,3}\|/) next
      spaces = 0
      while (spaces < length(line) && substr(line, spaces + 1, 1) == " ") spaces++
      sub(/^ {0,3}/, "", line)
      n = split(line, a, "|")
      id = trim(a[2])
      if (id ~ /^S[0-9]{3,}$/) printf "%s\tspaces=%d\tnf=%d\n", id, spaces, n
    }
  ' "$1"
}

extract_registry() {

    function trim(s) { gsub(/^[ \t]+|[ \t]+$/, "", s); return s }
    {
      line = $0
      if (line ~ /^ {0,3}##[ \t]/) {
        sub(/^ {0,3}/, "", line)
        in_registry = (trim(substr(line, 3)) == "Registry")
        next
      }
      if (line ~ /^ {0,3}#[ \t]/) { in_registry = 0; next }
      if (line !~ /^ {0,3}\|/) next

      n = split(line, a, "|")
      id = trim(a[2])
      if (id ~ /^S[0-9]{3,}$/) print id
    }
  ' "$1"
}
