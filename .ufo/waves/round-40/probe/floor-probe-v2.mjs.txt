// ---- embedded parser (copied from scripts/anchor-truth.mjs, same semantics) ----
const SPAN = /`([^`\n]+)`/gu;
const ANCH = /^([\p{L}\p{N}_./@+-]*[\p{L}\p{N}_-]\.[\p{L}\p{N}_]+)?:(\d+)(?:-(\d+))?$/u;
const PATHRE = /^[\p{L}\p{N}_./@+-]+\.(?:mjs|cjs|js|ts|json|md|sh|rs|toml|py|service|timer|ya?ml|txt|lock|conf)$/u;
const IDENT = /[A-Za-z_$][\p{L}\p{N}_$]*/gu;
const JOIN = /^[\s,;/()]*(?:(?:and|or)[\s,;/()]*)?$/u;
const SENT_END = /(?:\.\s+(?=[A-Z*`(])|\.\s*$|;\s|:\s*$)/u;
const strip = (s) => s.replace(/^[\s]+|[\s]+$/gu, '');
function* blocks(text) {
  let cur = [];
  let start = null;
  const lines = text.split('\n');
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    const ln = i + 1;
    if (strip(line) === '' || line.startsWith('#') || line.replace(/^\s+/u, '').startsWith('|') || line.startsWith('```')) {
      if (cur.length) { yield [cur.join(' '), start]; cur = []; }
      if (strip(line)) yield [line, ln];
      continue;
    }
    if (!cur.length) start = ln;
    cur.push(line);
  }
  if (cur.length) yield [cur.join(' '), start];
}
function parseNote(text) {
  const out = [];
  let curPath = null;
  let inCode = false;
  for (const [blk, ln] of blocks(text)) {
    if (blk.startsWith('```')) { inCode = !inCode; continue; }
    if (inCode) continue;
    if (blk.startsWith('#')) {
      const m = [...blk.matchAll(SPAN)].map((x) => x[1]).filter((s) => PATHRE.test(s) && s.includes('/') && !s.startsWith('.'));
      if (m.length) curPath = m[0];
    }
    const isRow = blk.replace(/^\s+/u, '').startsWith('|');
    const sentences = [];
    let sent = [];
    let pos = 0;
    for (const m of blk.matchAll(SPAN)) {
      const gap = blk.slice(pos, m.index);
      if (SENT_END.test(gap) || (isRow && gap.includes('|'))) {
        if (sent.length) sentences.push(sent);
        sent = [];
      }
      sent.push([m[1], gap, pos, m.index]);
      pos = m.index + m[0].length;
    }
    if (sent.length) sentences.push(sent);
    sentences.forEach((s) => {
      const sentence = blk;
      const items = s.map(([val, gap]) => {
        const am = ANCH.exec(val);
        if (am) return { kind: 'A', val, gap, path: am[1] ?? null, a: Number(am[2]), b: Number(am[3] ?? am[2]) };
        if (PATHRE.test(val)) return { kind: 'P', val, gap };
        return { kind: 'S', val, gap };
      });
      let pending = [];
      let i = 0;
      while (i < items.length) {
        const it = items[i];
        if (it.kind === 'P') {
          if (it.val.includes('/') && !it.val.startsWith('.')) curPath = it.val;
          i++; continue;
        }
        if (it.kind === 'S') { pending.push(it.val); i++; continue; }
        const run = [it];
        let j = i + 1;
        while (j < items.length && items[j].kind === 'A' && JOIN.test(items[j].gap)) { run.push(items[j]); j++; }
        const k = run.length;
        const syms = pending.length ? pending.slice(-k) : [];
        let nxt = null;
        if (syms.length === 0) {
          for (let q = j; q < items.length; q++) {
            if (items[q].kind !== 'S') continue;
            let r2 = q + 1;
            while (r2 < items.length && items[r2].kind === 'S' && JOIN.test(items[r2].gap)) r2++;
            const owned = r2 < items.length && items[r2].kind === 'A';
            if (!owned) nxt = items[q].val;
            break;
          }
        }
        const assign = syms.length ? [...Array(k - syms.length).fill(syms[0]), ...syms] : Array(k).fill(null);
        run.forEach((r, idx) => {
          const path = r.path ?? curPath;
          if (r.path) curPath = r.path;
          const a = assign[idx];
          out.push({ line: ln, path, a: r.a, b: r.b, sym: a !== null ? a : nxt, raw: r.val, sentence });
        });
        pending = [];
        i = j;
      }
    });
  }
  return out;
}
function norm(sym) {
  let s = strip(sym).replace(/^["']+|["']+$/g, '');
  s = s.replace(/\(.*\)$/u, '');
  s = s.replace(/^[.#-]+/u, '');
  const toks = s.match(IDENT) ?? [];
  let longest = s;
  if (toks.length) { longest = toks[0]; for (const t of toks) if (t.length > longest.length) longest = t; }
  return [s, longest];
}

function judge(path, a, b, token) {
  if (token == null) return 'unchecked';
  if (String(token).length > 256) return 'false';
  let rp = modFiles.includes(path) ? path : null;
  if (!rp && path) { const c = modFiles.filter((f) => f.endsWith('/' + path)); rp = c.length === 1 ? c[0] : null; }
  if (!rp) return 'dead';
  if (!modCache.has(rp)) { const r = git(src, ['show', `${srcRev}:${rp}`]); modCache.set(rp, r.status === 0 ? r.stdout.split('\n') : null); }
  const L = modCache.get(rp);
  if (L === null || a < 1 || b > L.length || a > b) return 'dead';
  const seg = L.slice(a - 1, b).join('\n');
  const [f, l] = norm(token);
  return (f && wb(f, seg)) || (l && wb(l, seg)) ? 'true' : 'false';
}

// ---- text span scanner and swap/reorder machinery (ported logic) ----
function scanSpans(text) {
  const out = []; let inCode = false; let pos = 0;
  for (const line of text.split('\n')) {
    if (line.startsWith('```')) inCode = !inCode;
    else if (!inCode) for (const m of line.matchAll(/`([^`\n]+)`/gu)) out.push([m[1], pos + m.index]);
    pos += line.length + 1;
  }
  return out;
}
function anchorSpanIndices(text) {
  const spans = scanSpans(text); const idx = [];
  spans.forEach(([v], i) => { if (ANCH.test(v)) idx.push(i); });
  return { spans, idx };
}
function offsetOf(anchors, spans, idx, ord) {
  const raw = anchors[ord].raw; const k = anchors.slice(0, ord).filter((x) => x.raw === raw).length;
  let seen = 0;
  for (const s of idx) { if (spans[s][0] === raw) { if (seen === k) return spans[s][1]; seen++; } }
  return null;
}
function swapBuild(note, i, j, mode) {
  const noteText = rd(note); const anchors = parseNote(noteText);
  const { spans, idx } = anchorSpanIndices(noteText);
  const oi = offsetOf(anchors, spans, idx, i); const oj = offsetOf(anchors, spans, idx, j);
  if (oi == null || oj == null) return { error: `offset missing i=${i} j=${j}` };
  const ri = anchors[i].raw; const rj = anchors[j].raw;
  const coord = (raw) => raw.slice(raw.lastIndexOf(':') + 1);
  const prefix = (raw) => raw.slice(0, raw.lastIndexOf(':') + 1);
  const ni = prefix(ri) + coord(rj); const nj = prefix(rj) + coord(ri);
  const edits = [[oi, ri, ni], [oj, rj, nj]].sort((x, y) => y[0] - x[0]);
  let t = noteText;
  for (const [o, r, n] of edits) t = t.slice(0, o + 1) + n + t.slice(o + 1 + r.length);
  const A2 = parseNote(t);
  const want = anchors.map((x, q) => (q === i ? ni : q === j ? nj : x.raw));
  if (A2.length !== anchors.length || A2.some((x, q) => x.raw !== want[q] || x.path !== anchors[q].path)) {
    return { error: `reparse mismatch i=${i} j=${j}` };
  }
  const p = pinOf(); const E = p.notes[note].entries;
  const old = [{ ...E[i] }, { ...E[j] }];
  for (const q of [i, j]) {
    const k = q === i ? 0 : 1; const x = A2[q]; const e = E[q];
    Object.assign(e, { raw: x.raw, a: x.a, b: x.b });
    if (mode === 'review-swap' || mode === 'pair-keep') { e.token = old[1 - k].token; e.source = 'review:probe'; }
    else { e.token = x.sym ?? null; e.source = 'pair'; }
    e.expect = ex(judge(e.path, e.a, e.b, e.token));
  }
  wr(note, t); writePin(p);
  return { ok: true, A2 };
}
