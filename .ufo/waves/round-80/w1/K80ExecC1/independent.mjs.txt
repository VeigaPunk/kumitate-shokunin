#!/usr/bin/env node
// Independent round-80 implementation: coverage and map are separate readings.
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync, readdirSync, mkdirSync, mkdtempSync, rmSync, lstatSync, renameSync, existsSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { tmpdir } from 'node:os';
import { fileURLToPath } from 'node:url';
const SELF = fileURLToPath(import.meta.url);
const ORACLE = join(dirname(SELF), 'readme-map.mjs');
const hash = b => createHash('sha256').update(b).digest('hex');
const README = 'corpus/README.md', PIN = '.ufo/pins/readme-map.json';
const rowKeys = text => text.split('\n').flatMap(line => {
  const m = line.match(/^ {0,3}\|\s*`((?:knowledge|lessons)\/[^`]+)`\s*\|/);
  return m ? [m[1]] : [];
});
function oracle(lane) {
  const r = spawnSync(process.execPath, [ORACLE, '--registry', join(lane, PIN), '--readme', join(lane, README), '--corpus', join(lane, 'corpus/knowledge'), '--corpus', join(lane, 'corpus/lessons')], { encoding: 'utf8' });
  const lines = (r.stdout || '').trim().split('\n');
  if (r.status === 0 && lines.some(l => l.startsWith('README_MAP: CLEAN '))) return { findings: [], first: null };
  const verdict = lines.find(l => l.startsWith('README_MAP: DRIFT '));
  const first = verdict?.match(/ first=(.*)$/)?.[1];
  if (r.status !== 1 || !first) throw Error('oracle');
  const findings = [first, ...lines.filter(l => l.startsWith('README_MAP: note ')).map(l => l.slice(17))];
  if (findings.some(f => f.startsWith('format-unmapped:'))) throw Error('oracle');
  return { findings, first };
}
const mapLine = (prefix, o) => `${prefix}: ${o.first ? `DRIFT first=${o.first}` : 'CLEAN'}`;
function recover(lane) {
  const root = join(lane, '.ufo/pins/adopt-journal');
  if (!existsSync(root)) return;
  for (const name of readdirSync(root).sort().filter(n => !n.endsWith('.retired'))) {
    const dir = join(root, name), m = JSON.parse(readFileSync(join(dir, 'manifest.json'), 'utf8'));
    if (m.contract !== 'kumitate-adopt-journal-v1' || m.lane !== lane || !Array.isArray(m.pre) || m.pre.length !== 2) throw Error('journal');
    const bytes = m.pre.map(p => {
      if (![README, PIN].includes(p.path) || !['README.pre', 'readme-map.pre'].includes(p.file)) throw Error('journal');
      const b = readFileSync(join(dir, p.file));
      if (hash(b) !== p.sha256) throw Error('journal');
      return b;
    });
    if (new Set(m.pre.map(p => p.path)).size !== 2) throw Error('journal');
    m.pre.forEach((p, i) => {
      const path = join(lane, p.path);
      if (!existsSync(path) || hash(readFileSync(path)) !== p.sha256) writeFileSync(path, bytes[i]);
    });
    renameSync(dir, `${dir}.retired`);
  }
}
function journal(lane, target, key) {
  const root = join(lane, '.ufo/pins/adopt-journal');
  mkdirSync(root, { recursive: true });
  const dir = mkdtempSync(join(root, 'adopt-'));
  const pre = [README, PIN].map((path, i) => {
    const file = i ? 'readme-map.pre' : 'README.pre', bytes = readFileSync(join(lane, path));
    writeFileSync(join(dir, file), bytes);
    return { path, file, sha256: hash(bytes) };
  });
  writeFileSync(join(dir, 'manifest.json'), JSON.stringify({ contract: 'kumitate-adopt-journal-v1', lane, target, key, pre }));
  return dir;
}
function execute(lane, mode, target) {
  recover(lane);
  const before = oracle(lane);
  const raw = readFileSync(join(lane, README), 'utf8');
  const pin = JSON.parse(readFileSync(join(lane, PIN), 'utf8'));
  const rows = rowKeys(raw), covered = key => rows.includes(key) && pin.covered.includes(key);
  if (mode === 'check') {
    const spots = readdirSync(join(lane, 'corpus/lessons')).filter(n => n.endsWith('.md') && n.includes('spotcheck')).sort();
    const uncovered = spots.filter(n => !covered(`lessons/${n}`));
    console.log(`SPOTCHECK_COVERAGE: ${uncovered.length ? 'DRIFT' : 'CLEAN'} spotchecks=${spots.length} uncovered=${uncovered.length}${uncovered.length ? ` first=corpus/lessons/${uncovered[0]}` : ''}`);
    console.log(mapLine('MAP', before));
    return uncovered.length ? 1 : 0;
  }
  const key = target.replace(/^corpus\//, '');
  if (!/^(knowledge|lessons)\/[A-Za-z0-9._-]+\.md$/.test(key) || !lstatSync(join(lane, 'corpus', key)).isFile()) throw Error('target');
  if (before.findings.some(f => f.startsWith('duplicate-row:'))) throw Error('oracle');
  if (covered(key)) { console.log(`SPOTCHECK_ADOPT: ALREADY_ADOPTED ${target}`); return 0; }
  const needRow = !rows.includes(key), needPin = !pin.covered.includes(key);
  const lines = raw.split('\n');
  if (needRow) {
    const start = lines.findIndex(l => /^##\s+Coverage map\s*$/.test(l));
    if (start < 0) throw Error('section');
    let insert = start + 1;
    for (let i = start + 1; i < lines.length && !/^##\s/.test(lines[i]); i++) if (/^\s*\|/.test(lines[i])) insert = i + 1;
    lines.splice(insert, 0, `| \`${key}\` | Verifier spotcheck adopted into the coverage map | ${target} |`);
  }
  if (needPin) pin.covered.push(key);
  const dir = journal(lane, target, key);
  try {
    if (needRow) writeFileSync(join(lane, README), lines.join('\n'));
    if (needPin) writeFileSync(join(lane, PIN), `${JSON.stringify(pin, null, 2)}\n`);
    const after = oracle(lane);
    const own = f => f.slice(f.indexOf(':') + 1) === key;
    if (!coveredAfter(lane, key) || after.findings.some(own) || JSON.stringify(before.findings.filter(f => !own(f))) !== JSON.stringify(after.findings.filter(f => !own(f)))) throw Error('delta');
    renameSync(dir, `${dir}.retired`);
    console.log(`SPOTCHECK_ADOPT: ADOPTED ${target} wrote=${needRow && needPin ? 'both' : needRow ? 'readme' : 'registry'}`);
    console.log(mapLine('README_MAP', after));
    return 0;
  } catch (e) { recover(lane); throw e; }
}
function coveredAfter(lane, key) {
  return rowKeys(readFileSync(join(lane, README), 'utf8')).includes(key) && JSON.parse(readFileSync(join(lane, PIN), 'utf8')).covered.includes(key);
}
function selftest() {
  const root = mkdtempSync(join(tmpdir(), 'independent-c1-'));
  let n = 0, pass = 0;
  const check = (name, ok) => { n++; if (ok) pass++; console.log(`case ${name}: ${ok ? 'PASS' : 'FAIL'}`); };
  const target = 'corpus/lessons/round-75-verifier-spotcheck.md', key = target.slice(7);
  const fixture = (name, adopted = false) => {
    const d = join(root, name);
    for (const p of ['corpus/knowledge', 'corpus/lessons', '.ufo/pins']) mkdirSync(join(d,p), {recursive:true});
    writeFileSync(join(d,'corpus/knowledge/base.md'),'# Base\n');
    writeFileSync(join(d,target),'# Spotcheck\n');
    writeFileSync(join(d,README), `# Corpus\n\n## Coverage map\n\n| \`knowledge/base.md\` | Base | fixture |\n${adopted ? `| \`${key}\` | Spotcheck | fixture |\n` : ''}\n## Known gaps\n\n- none\n`);
    writeFileSync(join(d,PIN), `${JSON.stringify({version:2, covered:['knowledge/base.md', ...(adopted ? [key] : [])]},null,2)}\n`);
    return d;
  };
  const run = (d,args,extra={}) => spawnSync(process.execPath,[SELF,'--lane',d,...args], {encoding:'utf8', ...extra});
  const bytes = d => [README,PIN].map(p => hash(readFileSync(join(d,p)))).join(':');
  try {
    let d=fixture('clean',true), r=run(d,['--check']);
    check('clean-exit',r.status===0); check('clean-coverage',r.stdout.includes('SPOTCHECK_COVERAGE: CLEAN spotchecks=1 uncovered=0')); check('clean-map',r.stdout.includes('MAP: CLEAN'));
    d=fixture('uncovered'); r=run(d,['--check']);
    check('uncovered-exit',r.status===1); check('uncovered-name',r.stdout.includes(`uncovered=1 first=${target}`)); check('uncovered-map',r.stdout.includes('MAP: DRIFT first='));
    d=fixture('map-only',true); writeFileSync(join(d,'corpus/knowledge/late.md'),'# Late\n'); r=run(d,['--check']);
    check('map-only-exit',r.status===0); check('map-only-coverage',r.stdout.includes('CLEAN spotchecks=1 uncovered=0')); check('map-only-first',r.stdout.includes('MAP: DRIFT first=uncovered:knowledge/late.md'));
    d=fixture('orphan',true); writeFileSync(join(d,README),readFileSync(join(d,README),'utf8')+'| `knowledge/ghost.md` | Ghost | fixture |\n'); r=run(d,['--check']);
    check('orphan-exit',r.status===0); check('orphan-first',r.stdout.includes('MAP: DRIFT first=orphan-row:knowledge/ghost.md'));
    d=fixture('format',true); writeFileSync(join(d,README),'# Corpus\n| lessons/not-a-row | broken |\n'); r=run(d,['--check']);
    check('format-refused',r.status===2 && r.stdout.trim()==='REFUSED_ORACLE');
    r=run(d,['--bogus']); check('usage',r.status===64 && r.stdout.trim()==='SPOTCHECK_ADOPT: USAGE');
    d=fixture('adopt'); writeFileSync(join(d,'corpus/knowledge/late.md'),'# Late\n'); r=run(d,['--adopt',target]);
    check('adopt-exit',r.status===0); check('adopt-both',r.stdout.startsWith(`SPOTCHECK_ADOPT: ADOPTED ${target} wrote=both\n`)); check('adopt-red-map',r.stdout.includes('README_MAP: DRIFT first=uncovered:knowledge/late.md')); check('adopt-row',rowKeys(readFileSync(join(d,README),'utf8')).includes(key)); check('adopt-pin',JSON.parse(readFileSync(join(d,PIN),'utf8')).covered.includes(key));
    const before=bytes(d); r=run(d,['--adopt',target]); check('idempotent-exit',r.status===0); check('idempotent-line',r.stdout.trim()===`SPOTCHECK_ADOPT: ALREADY_ADOPTED ${target}`); check('idempotent-bytes',bytes(d)===before);
    d=fixture('duplicate',true); writeFileSync(join(d,README),readFileSync(join(d,README),'utf8')+`| \`${key}\` | Again | fixture |\n`); const dup=bytes(d); r=run(d,['--adopt',target]); check('duplicate-refused',r.status===2); check('duplicate-bytes',bytes(d)===dup);
    d=fixture('corrupt'); const j=journal(d,target,key); writeFileSync(join(j,'README.pre'),'corrupt'); const corrupt=bytes(d); r=run(d,['--check']); check('corrupt-refused',r.status===2 && r.stdout.trim()==='REFUSED_ORACLE'); check('corrupt-bytes',bytes(d)===corrupt);
    d=fixture('recovery'); const pre=bytes(d), valid=journal(d,target,key); writeFileSync(join(d,README),'interrupted'); r=run(d,['--check']); check('recovery-bytes',bytes(d)===pre); check('recovery-retired',existsSync(`${valid}.retired`)); check('recovery-reading',r.status===1);
    d=fixture('invalid'); writeFileSync(join(d,PIN),'invalid'); r=run(d,['--check']); check('invalid-registry',r.status===2 && r.stdout.trim()==='REFUSED_ORACLE');
    d=fixture('happy'); r=run(d,['--adopt',target]); check('happy-adopt',r.status===0); r=run(d,['--check']); check('happy-check',r.status===0 && r.stdout.includes('MAP: CLEAN'));
  } finally { rmSync(root,{recursive:true,force:true}); }
  console.log(`SPOTCHECK_ADOPT_SELFTEST: ${pass===n?'PASS':'FAIL'} (${pass}/${n})`);
  return pass===n?0:1;
}
function main(argv) {
  let lane=process.cwd(), mode=null, target;
  for(let i=0;i<argv.length;i++) {
    const a=argv[i];
    if(a==='--lane' && argv[i+1]) lane=resolve(argv[++i]);
    else if(a==='--check' && !mode) mode='check';
    else if(a==='--adopt' && !mode && argv[i+1]) {mode='adopt';target=argv[++i];}
    else if(a==='--selftest' && !mode) mode='selftest';
    else { console.log('SPOTCHECK_ADOPT: USAGE'); return 64; }
  }
  if(!mode) {console.log('SPOTCHECK_ADOPT: USAGE');return 64;}
  try {return mode==='selftest'?selftest():execute(lane,mode,target);}
  catch {console.log('REFUSED_ORACLE');return 2;}
}
process.exitCode=main(process.argv.slice(2));
